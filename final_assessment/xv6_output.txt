# Mission 22 xv6 experiment — genuine run, macOS host, QEMU 11.1.3 (riscv64-softmmu),
# xv6-riscv (MIT, upstream main as of 2026-11-25), riscv64-elf-gcc cross-compiler.
#
# Build:
#   (apt/brew) riscv64-elf-gcc, riscv64-elf-binutils, qemu-system-riscv64
#   make                 # builds kernel/kernel + fs.img including stackbomb
#   make qemu            # boots xv6 under QEMU, -nographic serial console
#
# At the xv6 shell prompt:
#   stackbomb

qemu-system-riscv64 -machine virt -bios none -kernel kernel/kernel -m 128M -smp 3 -nographic -global virtio-mmio.force-legacy=false -drive file=fs.img,if=none,format=raw,id=x0 -device virtio-blk-device,drive=x0,bus=virtio-mmio-bus.0
